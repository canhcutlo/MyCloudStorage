import joblib
import os
from huggingface_hub import hf_hub_download

# Define a path to store the downloaded model
model_dir = './model_files'
os.makedirs(model_dir, exist_ok=True)

# Fetch the model from Hugging Face Hub
model_path = hf_hub_download(repo_id="saifhmb/fraud-detection-model", filename="skops-jbninwmt.pkl", local_dir=model_dir, local_dir_use_symlinks=False)

# Load the model
model = joblib.load(model_path)

# only load pickle files from sources you trust
# read more about it here https://skops.readthedocs.io/en/stable/persistence.html

import pandas as pd

# Create sample data similar to what a fraud detection model might expect
# (These are example features, replace with actual features if known)
sample_data = pd.DataFrame({
    'step': [1, 10, 20, 30, 40],
    'type': ['PAYMENT', 'TRANSFER', 'CASH_OUT', 'DEBIT', 'CASH_IN'],
    'amount': [1000.0, 5000.0, 200.0, 50.0, 1500.0],
    'nameOrig': ['C123', 'C456', 'C789', 'C101', 'C112'],
    'oldbalanceOrg': [10000.0, 15000.0, 500.0, 100.0, 2000.0],
    'newbalanceOrg': [9000.0, 10000.0, 300.0, 50.0, 3500.0],
    'nameDest': ['M987', 'C654', 'M321', 'C212', 'M432'],
    'oldbalanceDest': [5000.0, 2000.0, 1000.0, 200.0, 500.0],
    'newbalanceDest': [6000.0, 7000.0, 1200.0, 250.0, 2000.0],
    'Bene_Country': ['US', 'CA', 'US', 'MX', 'UK'], # Added missing column
    'Transaction_Type': ['Online', 'ATM', 'POS', 'Online', 'ATM'], # Added missing column
    'USD_amount': [1000.0, 5000.0, 200.0, 50.0, 1500.0], # Added missing column, assuming it's same as 'amount'
    'Sender_Country': ['US', 'US', 'US', 'US', 'US'] # Added missing column
})

print("Dữ liệu mẫu:")
display(sample_data)

# Make predictions using the loaded model
# The model is a sklearn pipeline, so it should handle preprocessing
predictions = model.predict(sample_data)

print("\nDự đoán gian lận (0 = không gian lận, 1 = gian lận):")
display(predictions)
